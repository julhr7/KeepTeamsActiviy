' Script para mantener Teams "Activo" (Estado Verde)
' Guardar como KeepTeamsActive.vbs y ejecutar con doble clic

Set objShell = CreateObject("WScript.Shell")
Set objFSO = CreateObject("Scripting.FileSystemObject")

' Ruta del archivo de registro
logFolder = "C:\Temp\"
logFile = logFolder & "TeamsActivityLog.txt"

' Crear carpeta si no existe
If Not objFSO.FolderExists(logFolder) Then
    objFSO.CreateFolder(logFolder)
End If

' Función para simular actividad
Sub SimulateActivity()
    ' Presionar tecla Shift (sin efecto visible)
    objShell.SendKeys("{Shift}")
    
    ' Registrar actividad (modo correcto para abrir archivo)
    Set file = objFSO.OpenTextFile(logFile, 8, True)
    file.WriteLine "Actividad simulada: " & Now()
    file.Close
End Sub

' Mensaje inicial
MsgBox "Script iniciado. Teams permanecerá en estado 'Activo'.", vbInformation, "Teams Activity"

' Bucle principal cada 5 minutos
On Error Resume Next  ' Para evitar que el script se detenga por errores menores
Do While True
    SimulateActivity
    WScript.Sleep(300000)  ' 300 segundos = 5 minutos
Loop
